interface ContactRecord {
  id: string;
  email: string;
}
interface EventRecord {
  id: string;
  acceptance_id: string;
  kind: "accepted_new" | "resource_open";
}
const contactTable = defineTable<ContactRecord, "id">({
  name: "repro_contact", primaryKey: "id", columns: ["id", "email"],
});
const eventTable = defineTable<EventRecord, "id">({
  name: "repro_event", primaryKey: "id", columns: ["id", "acceptance_id", "kind"],
});
class Contact extends defineModel(contactTable) {
  events() {
    return this.hasMany<EventRecord, "id">(() => Event, "acceptance_id", "id");
  }
}
class Event extends defineModel(eventTable) {
  contact() {
    return this.belongsTo<ContactRecord, "id">(() => Contact, "acceptance_id", "id");
  }
}
